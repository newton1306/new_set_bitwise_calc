    Start([Start])

    PrintHeader --> CheckEmpty{nsets == 0?}

    PrintSet --> Inc[Increment i]
    Inc --> Loop
    Loop -->|No| PrintDiv[Print divider]
    PrintNoSets --> End([End])
    PrintDiv --> End
